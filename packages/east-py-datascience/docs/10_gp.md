# Module 10: Gaussian Process (`gp_impl.py`)

## Purpose

Gaussian Process regression for probabilistic predictions with uncertainty.

Note: GPflow models cannot be exported to ONNX, so we use native cloudpickle serialization.

## Config Types

```python
GPConfigType = StructType([
    ("kernel", OptionType(GPKernelType)),        # default rbf
    ("noise_variance", OptionType(FloatType)),   # default 0.1
    ("max_iter", OptionType(IntegerType)),       # default 1000
])

GPPredictConfigType = StructType([
    ("confidence_level", OptionType(FloatType)),  # default 0.95
])
```

## Platform Functions

### `gp_train`

Train Gaussian Process model.

```python
PlatformFunction(
    name="gp_train",
    inputs=[MatrixType, VectorType, GPConfigType],
    output=ModelBlobType,  # Returns "gp_regressor" variant
    type="sync",
    fn=gp_train_impl,
)

def gp_train_impl(
    X: Matrix,
    y: Vector,
    config: EastStruct
) -> ModelBlob:
    """Train Gaussian Process model using GPflow."""
    import numpy as np
    import gpflow

    X_np = east_matrix_to_numpy(X).astype(np.float64)
    y_np = east_vector_to_numpy(y).astype(np.float64).reshape(-1, 1)
    n_features = X_np.shape[1]

    kernel_variant = _get_option(config.get("kernel"), None)
    kernel_name = _get_enum_tag(kernel_variant) if kernel_variant else "rbf"

    kernel_map = {
        "rbf": gpflow.kernels.SquaredExponential,
        "matern32": gpflow.kernels.Matern32,
        "matern52": gpflow.kernels.Matern52,
        "linear": gpflow.kernels.Linear,
    }
    kernel_cls = kernel_map.get(kernel_name, gpflow.kernels.SquaredExponential)
    kernel = kernel_cls()

    noise_variance = _get_option(config.get("noise_variance"), 0.1)

    model = gpflow.models.GPR(
        data=(X_np, y_np),
        kernel=kernel,
        noise_variance=noise_variance,
    )

    # Optimize hyperparameters
    max_iter = _get_option(config.get("max_iter"), 1000)
    opt = gpflow.optimizers.Scipy()
    opt.minimize(
        model.training_loss,
        model.trainable_variables,
        options={"maxiter": max_iter},
    )

    return EastVariant("gp_regressor", EastStruct({
        "data": _serialize_native(model),
        "kernel": _make_enum(kernel_name),
        "n_features": n_features,
    }))
```

### `gp_predict`

Get predictions with uncertainty.

```python
PlatformFunction(
    name="gp_predict",
    inputs=[ModelBlobType, MatrixType, GPPredictConfigType],
    output=PredictionResultType,
    type="sync",
    fn=gp_predict_impl,
)

def gp_predict_impl(
    model_blob: ModelBlob,
    X: Matrix,
    config: EastStruct
) -> EastStruct:
    """Predict with Gaussian Process model."""
    import numpy as np
    from scipy import stats

    if model_blob.type != "gp_regressor":
        raise ValueError(f"Expected gp_regressor, got {model_blob.type}")

    model = _deserialize_native(model_blob.value["data"])
    X_np = east_matrix_to_numpy(X).astype(np.float64)

    mean, var = model.predict_f(X_np)
    mean_np = mean.numpy().flatten()
    std_np = np.sqrt(var.numpy()).flatten()

    confidence = _get_option(config.get("confidence_level"), 0.95)
    z = stats.norm.ppf(1 - (1 - confidence) / 2)

    lower = mean_np - z * std_np
    upper = mean_np + z * std_np

    return EastStruct({
        "predictions": EastArray(FloatType, [float(m) for m in mean_np]),
        "std": EastSome(EastArray(FloatType, [float(s) for s in std_np])),
        "lower": EastSome(EastArray(FloatType, [float(l) for l in lower])),
        "upper": EastSome(EastArray(FloatType, [float(u) for u in upper])),
    })
```

### `gp_predict_point`

Get point predictions only (no uncertainty).

```python
PlatformFunction(
    name="gp_predict_point",
    inputs=[ModelBlobType, MatrixType],
    output=VectorType,
    type="sync",
    fn=gp_predict_point_impl,
)

def gp_predict_point_impl(
    model_blob: ModelBlob,
    X: Matrix
) -> Vector:
    """Get point predictions from Gaussian Process model."""
    import numpy as np

    if model_blob.type != "gp_regressor":
        raise ValueError(f"Expected gp_regressor, got {model_blob.type}")

    model = _deserialize_native(model_blob.value["data"])
    X_np = east_matrix_to_numpy(X).astype(np.float64)

    mean, _ = model.predict_f(X_np)
    mean_np = mean.numpy().flatten()

    return EastArray(FloatType, [float(m) for m in mean_np])
```
